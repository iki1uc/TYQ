// ============================================================ // EVO · TMP-ENGINE · Universell · Entkoppelt // ============================================================ // // Aufgabe: // - Evolution dynamisch berechnen // - keine festen Stufen // - keine festen Symbole // - alles TMP geführt // - Werte dynamisch sortieren (3, 9, 81, 756, 27) // - keine Rückverfolgung // ============================================================ const EVO = { tmp: {}, sorted: {}, evoLevel: 0, init() { this.resetTMP(); this.loadBase(); this.sortValues(); this.calcEvolution(); this.updateUI(); }, resetTMP() { this.tmp = {}; this.sorted = {}; console.log("🌀 EVO TMP: Speicher geleert"); }, loadBase() { // Dynamisch, nicht fest this.tmp.base = [3, 9, 81, 756, 27]; console.log("🌀 EVO TMP: Basiswerte geladen"); }, sortValues() { const base = this.tmp.base; this.sorted = { numbers: base, letters: ["A", "B", "C", "D", "E"], percent: base.map(v => v + "%"), degree: base.map(v => v + "°"), te: base.map(v => v + "te") }; console.log("🌀 EVO TMP: Werte sortiert"); }, calcEvolution() { // Dynamische Evolution: // Anzahl der gültigen Werte bestimmt die Stufe const valid = this.sorted.numbers.filter(v => v > 0).length; this.evoLevel = valid; // 1–5 dynamisch console.log("🌀 EVO TMP: Evolution =", this.evoLevel); }, updateUI() { const el = document.getElementById("evoLevel"); if (el) el.textContent = this.evoLevel; } }; window.EVO = EVO; EVO.init();
